cmake_minimum_required(VERSION 3.16)
project(LocalArcade LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Snake and Tetris console games currently require Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(arcade Project/main.cpp)
target_link_libraries(arcade PRIVATE bcrypt)
if(MSVC)
  target_compile_definitions(arcade PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
endif()
enable_testing()
add_executable(arcade_tests tests/arcade_tests.cpp)
target_link_libraries(arcade_tests PRIVATE bcrypt)
if(MSVC)
  target_compile_definitions(arcade_tests PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
endif()
add_test(NAME arcade_regressions COMMAND arcade_tests)
